 const header=page.getByRole('button',{name:'Arrival checklist / collapsed'});await header.focus();await page.keyboard.press('Enter');await expect(page.getByRole('button',{name:'Arrival checklist / expanded'})).toHaveAttribute('aria-expanded','true');checks.push('Accordion.Item scoped text/selected slot responds to keyboard');

 for(let i=0;i<3;i++){await page.getByRole('button',{name:'Add one seat',exact:true}).click();await expect(page.getByTestId('seats')).toHaveText('Extra seats: 1');await page.getByRole('button',{name:'Show or hide seat counter',exact:true}).click();await expect(page.getByTestId('cleanup')).toHaveText(`Cleanups: ${i+1}`);await page.getByRole('button',{name:'Show or hide seat counter',exact:true}).click();await expect(page.getByTestId('seats')).toHaveText('Extra seats: 0');}
 checks.push('raw per-instance ctrl subscription: three mount/action/unmount/remount cycles; one event per action, reset on remount, cleanup called');
 expect(diagnostics).toEqual([]);results.runs.push({name,status:'passed',checks,diagnostics});
 }catch(e){results.runs.push({name,status:'failed',checks,error:e.message,diagnostics});await page.screenshot({path:`../${name}-failure.png`});}
 await context.close();
}
