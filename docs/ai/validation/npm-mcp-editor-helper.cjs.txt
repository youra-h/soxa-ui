const fs=require('node:fs');const vscode=require('vscode');exports.activate=async()=>{const report=require('node:path').join(__dirname, 'editor-evidence.json');try{await new Promise(r=>setTimeout(r,4000));await vscode.commands.executeCommand('workbench.mcp.startServer','*',{waitForLiveTools:true});await new Promise(r=>setTimeout(r,1000));const tools=vscode.lm.tools.filter(t=>t.name.startsWith('mcp_soxa-ui-docs_'));const call=async(suffix,input)=>{const result=await vscode.lm.invokeTool(tools.find(t=>t.name.endsWith(suffix)).name,{input});return result.content.map(p=>p.value||'').join('');};const versions=await call('list_versions',{});const select=await call('get_component',{version:'0.4.2',audience:'consumer',framework:'vue',component:'select',offset:0,limit:1000});fs.writeFileSync(report,JSON.stringify({version:vscode.version,tools:tools.map(t=>t.name),versions,select,completed:true},null,2));}catch(error){fs.writeFileSync(report,JSON.stringify({error:String(error),stack:error.stack},null,2));}}
